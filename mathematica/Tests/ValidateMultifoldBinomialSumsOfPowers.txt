(* ================================================================ *)
(*  Validates Proposition: Multifold Binomial Sums of Powers        *)
(* ================================================================ *)

ClearAll[
  KnuthRFoldSum,
  ForwardDifference,
  MultifoldBinomialSumsOfPowers,
  ValidateMultifoldBinomialSumsOfPowers
];

(* --------------------------------------------------------------- *)
(*  Convention: x^0 = 1 for all x                                   *)
(* --------------------------------------------------------------- *)

Begin["`Private`"]

Unprotect[Power];
Power[_?NumericQ | _, 0] = 1;
Protect[Power];

End[];


(* --------------------------------------------------------------- *)
(*  Multifold sums of powers                                        *)
(* --------------------------------------------------------------- *)

KnuthRFoldSum[r_, n_, m_] := 0 /; n <= 0;

KnuthRFoldSum[0, n_, m_] :=
  n^m;

KnuthRFoldSum[r_, n_, m_] :=
  Sum[
    KnuthRFoldSum[r - 1, k, m],
    {k, 1, n}
  ] /; r > 0;


(* --------------------------------------------------------------- *)
(*  Forward difference operator                                     *)
(*                                                                   *)
(*      Δ^k t^m                                                     *)
(* --------------------------------------------------------------- *)

ForwardDifference[t_, m_, k_] :=
  Sum[
    (-1)^(k - j)
      Binomial[k, j]
      (t + j)^m,
    {j, 0, k}
  ];


(* --------------------------------------------------------------- *)
(*  Proposition                                                     *)
(*                                                                   *)
(*      Σ^r n^m                                                     *)
(*        = Σ                                                       *)
(*          [                                                       *)
(*            Binomial[n-t+r, k+r]                                  *)
(*            - Σ Binomial[r-s-t, k+r-s] Binomial[s+n-1,s]          *)
(*          ]                                                       *)
(*          Δ^k t^m                                                 *)
(* --------------------------------------------------------------- *)

MultifoldBinomialSumsOfPowers[r_, n_, m_, t_] :=
  Sum[
    (
      Binomial[n - t + r, k + r]

      -

      Sum[
        Binomial[r - s - t, k + r - s]
          *
        Binomial[s + n - 1, s],
        {s, 0, r - 1}
      ]
    )
    *
    ForwardDifference[t, m, k],
    {k, 0, m}
  ];


(* --------------------------------------------------------------- *)
(*  Validation routine                                              *)
(*                                                                   *)
(*  All entries should evaluate to 0                                *)
(* --------------------------------------------------------------- *)

ValidateMultifoldBinomialSumsOfPowers[max_] :=
  Flatten[
    Table[
      FullSimplify[
        KnuthRFoldSum[r, n, m]
          -
        MultifoldBinomialSumsOfPowers[r, n, m, t]
      ],
      {r, 0, max},
      {n, 1, max},
      {m, 0, max},
      {t, -max, max}
    ]
  ];


(* --------------------------------------------------------------- *)
(*  Run validation                                                  *)
(* --------------------------------------------------------------- *)

ValidateMultifoldBinomialSumsOfPowers[6]
