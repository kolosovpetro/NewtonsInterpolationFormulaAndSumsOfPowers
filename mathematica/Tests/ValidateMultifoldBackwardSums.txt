(* ================================================================ *)
(*  Validates Proposition: Multifold Backward Sums                  *)
(* ================================================================ *)

ClearAll[
  KnuthRFoldSum,
  EulerianNumber,
  BackwardDifference,
  MultifoldBackwardSums,
  ValidateMultifoldBackwardSums
];

(* --------------------------------------------------------------- *)
(*  Convention: x^0 = 1 for all x                                   *)
(* --------------------------------------------------------------- *)

Begin["`Private`"]

Unprotect[Power];
Power[_?NumericQ | _, 0] = 1;
Protect[Power];

End[];


(* --------------------------------------------------------------- *)
(*  Multifold sums of powers                                        *)
(* --------------------------------------------------------------- *)

KnuthRFoldSum[r_, n_, m_] := 0 /; n <= 0;

KnuthRFoldSum[0, n_, m_] := n^m;

KnuthRFoldSum[r_, n_, m_] :=
  Sum[
    KnuthRFoldSum[r - 1, k, m],
    {k, 1, n}
  ] /; r > 0;


(* --------------------------------------------------------------- *)
(*  Eulerian numbers (provided but not required here)               *)
(* --------------------------------------------------------------- *)

EulerianNumber[0, 0] = 1;
EulerianNumber[n_, k_] /; k < 0 || k >= n := 0;
EulerianNumber[n_, k_] :=
  EulerianNumber[n, k] =
    (k + 1) EulerianNumber[n - 1, k] +
    (n - k) EulerianNumber[n - 1, k - 1];


(* --------------------------------------------------------------- *)
(*  Backward difference operator                                    *)
(*      ∇^k (t+k)^m                                                 *)
(* --------------------------------------------------------------- *)

BackwardDifference[t_, m_, k_] :=
  Sum[
    (-1)^j Binomial[k, j] (t - j + k)^m,
    {j, 0, k}
  ];


(* --------------------------------------------------------------- *)
(*  Proposition                                                     *)
(*                                                                   *)
(*  Σ^r n^m = Σ                                                    *)
(*    [                                                            *)
(*      Binomial[n - t + r, k + r]                                 *)
(*      - Σ Binomial[r - s - t, k + r - s] Σ^s n^0                 *)
(*    ]                                                            *)
(*    ∇^k (t+k)^m                                                  *)
(* --------------------------------------------------------------- *)

MultifoldBackwardSums[r_, n_, m_, t_] :=
  Sum[
    (
      Binomial[n - t + r, k + r]
      -
      Sum[
        Binomial[r - s - t, k + r - s]
          *
        KnuthRFoldSum[s, n, 0],
        {s, 0, r - 1}
      ]
    )
    *
    BackwardDifference[t, m, k],
    {k, 0, m}
  ];


(* --------------------------------------------------------------- *)
(*  Validation routine                                              *)
(* --------------------------------------------------------------- *)

ValidateMultifoldBackwardSums[max_] :=
  Flatten[
    Table[
      FullSimplify[
        KnuthRFoldSum[r, n, m]
          -
        MultifoldBackwardSums[r, n, m, t]
      ],
      {r, 0, max},
      {n, 1, max},
      {m, 0, max},
      {t, -max, max}
    ]
  ];


(* --------------------------------------------------------------- *)
(*  Run validation                                                  *)
(* --------------------------------------------------------------- *)

ValidateMultifoldBackwardSums[5]
