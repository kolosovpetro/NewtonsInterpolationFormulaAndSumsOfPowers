Because
\begin{proposition}[Newton's interpolation formula for powers]
  \label{prop:newtons-interpolation-formula-for-powers}
  For non-negative integers $m,n$ and an arbitrary integer $t$
  \begin{align*}
    n^m = \sum_{k=0}^{m} \binom{n-t}{k} \Delta^{k} t^m
  \end{align*}
\end{proposition}
