\begin{lemma}[Stirling finite differences]
  \label{lem:finite-differences-via-stirling-numbers}
  For non-negative integers $j,m$, and an arbitrary integer $t$,
  \begin{align*}
    \Delta^{k} t^{m} = \sum_{s=0}^{m} \stirlingii{m}{k+s} \binom{t}{s} (k+s)!
  \end{align*}
  \begin{proof}
    By well-known identity 
    $t^{m} = \sum_{s=0}^{m} \stirlingii{m}{s} \binom{t}{s} s!$
  \end{proof}
\end{lemma}
