Forward finite differences of powers are closely related
to Stirling numbers of the second kind $\stirlingii{n}{k}$,
as the following well-known lemma shows.

\begin{lemma}[Stirling finite differences]
  \label{lem:finite-differences-via-stirling-numbers}
  For non-negative integers $j,m$, and an arbitrary integer $t$,
  \begin{align*}
    \Delta^{j} t^{m} = \sum_{k=0}^{m} \stirlingii{m}{j+k} \binom{t}{k} (j+k)!
  \end{align*}
\end{lemma}

This lemma
implies variations for ordinary sums of powers.
By utilizing Proposition~\eqref{prop:ordinary-sums-of-powers-via-newtons-formula},
and Lemma~\eqref{lem:finite-differences-via-stirling-numbers}, we obtain
formula for sums of powers
in terms of Stirling numbers of the second kind $\stirlingii{n}{k}$.

\begin{proposition}[Stirling sums of powers]
  \label{prop:ordinary-sums-of-powers-via-stirling-numbers}
  For integers $r\geq0$, and $n\geq 0$, and $m\geq 0$,
  and an arbitrary integer $t$,
  \begin{align*}
    \KnuthRFoldSum{1}{n}{m}
    = \sum_{j=0}^{m} \sum_{k=0}^{m}
    \left[ \binom{n-t+1}{j+1} - \binom{1-t}{j+1} \right]
    \stirlingii{m}{j+k} \binom{t}{k} (j+k)!
  \end{align*}
\end{proposition}

In addition, by using the negated formula for sums of
powers in
Proposition~\eqref{prop:negation-of-ordinary-sums-of-powers-via-newtons-formula},
with Lemma~\eqref{lem:finite-differences-via-stirling-numbers} for
finite differences in Stirling numbers $\stirlingii{n}{k}$,
we obtain new result.

\begin{proposition}[Negated Stirling sums of powers]
  \label{prop:negated-ordinary-sums-of-powers-via-stirling-numbers}
  For integers $r\geq0$, and $n\geq 0$, and $m\geq 0$,
  and an arbitrary integer $t$,
  \begin{align*}
    \KnuthRFoldSum{1}{n}{m}
    = \sum_{j=0}^{m} \sum_{k=0}^{m}
    \left[ \binom{n-t+1}{j+1} + (-1)^j \binom{j+t-1}{j+1} \right]
    \stirlingii{m}{j+k} \binom{t}{k} (j+k)!
  \end{align*}
\end{proposition}

By setting $t=0$ into the
Proposition~\eqref{prop:negated-ordinary-sums-of-powers-via-stirling-numbers}
yields well-known identity for sums of powers in terms of Stirling numbers.
That is,
\begin{corollary}
  For non-negative integers $n,m$,
  \begin{align*}
    \KnuthRFoldSum{1}{n}{m} = \sum_{j=0}^{m} \binom{n+1}{j+1} \stirlingii{m}{j} j!
  \end{align*}
\end{corollary}

Similarly,
formula for multifold sums of
powers
in terms of Stirling numbers of the second kind $\stirlingii{n}{k}$
follows.

\begin{proposition}[Multifold Stirling sums of powers]
  \label{prop:multifold-stirling-sums-of-powers}
  For integers $r\geq0$, and $n\geq 0$, and $m\geq 0$,
  and an arbitrary integer $t$,
  \begin{align*}
    \KnuthRFoldSum{r}{n}{m}
    = \sum_{j=0}^{m} \sum_{k=0}^{m}
    \left[ \binom{n-t+r}{j+r} - \sum_{s=1}^{r} \binom{s-t}{j+s} \KnuthRFoldSum{r-s}{n}{0} \right]
    \stirlingii{m}{j+k} \binom{t}{k} (j+k)!
  \end{align*}
\end{proposition}

By utilizing Lemma~\eqref{lem:negated-hockey-stick-identity} for
negated hockey stick identity, we obtain the negated formula for
multifold sums of powers in terms
of Stirling numbers of the second kind $\stirlingii{n}{k}$.

\begin{proposition}[Negated multifold Stirling sums of powers]
  \label{prop:multifold-sums-of-powers-via-stirling-numbers}
  For integers $r\geq0$, and $n\geq 0$, and $m\geq 0$,
  and an arbitrary integer $t$,
  \begin{align*}
    \KnuthRFoldSum{r}{n}{m}
    = \sum_{j=0}^{m} \sum_{k=0}^{m}
    \left[
      \tbinom{n-t+r}{j+r}
      + \sum_{s=1}^{r} (-1)^{j+s-1}
      \tbinom{j+t-1}{j+s} \KnuthRFoldSum{r-s}{n}{0}
    \right]
    \tstirlingii{m}{j+k} \tbinom{t}{k} (j+k)!
  \end{align*}
\end{proposition}

Propositions~\eqref{prop:multifold-stirling-sums-of-powers},
and~\eqref{prop:multifold-sums-of-powers-via-stirling-numbers}
can be expressed in a pure binomial form,
by means of the identity~\eqref{prop:multifold-sum-of-zero-powers},
which implies that
$\KnuthRFoldSum{r-s}{n}{0} = \binom{r-s+n-1}{r-s}$.
Therefore, we get multifold Stirling-Binomial formula for sums of powers.

\begin{proposition}[Multifold Stirling-Binomial sums of powers]
  % \label{prop:multifold-sums-of-powers-binomial-form}
  For integers $r\geq0$, and $n\geq 0$, and $m\geq 0$,
  and an arbitrary integer $t$,
  \begin{align*}
    \KnuthRFoldSum{r}{n}{m}
    = \sum_{j=0}^{m} \sum_{k=0}^{m}
    \left[ \binom{n-t+r}{j+r} - \sum_{s=1}^{r} \binom{s-t}{j+s} \binom{r-s+n-1}{r-s} \right]
    \stirlingii{m}{j+k} \binom{t}{k} (j+k)!
  \end{align*}
\end{proposition}
Lemma~\eqref{lem:negated-hockey-stick-identity} yields its negated version.
\begin{proposition}[Negated Stirling-Binomial sums of powers]
  % \label{prop:negated-multifold-sums-of-powers-binomial-form}
  For integers $r\geq0$, and $n\geq 0$, and $m\geq 0$,
  and an arbitrary integer $t$,
  \begin{align*}
    \KnuthRFoldSum{r}{n}{m}
    = \sum_{j=0}^{m} \sum_{k=0}^{m}
    \left[
      \tbinom{n-t+r}{j+r} + \sum_{s=1}^{r} (-1)^{j+s-1}
      \tbinom{j+t-1}{j+s} \tbinom{r-s+n-1}{r-s}
    \right]
    \tstirlingii{m}{j+k} \tbinom{t}{k} (j+k)!
  \end{align*}
\end{proposition}
