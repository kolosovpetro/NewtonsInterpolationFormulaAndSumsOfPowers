In this manuscript, we derive formulas for multifold sums of powers
by utilizing Newton's interpolation formula.
Furthermore, we provide formulas for multifold sums of powers in terms
of Stirling numbers of the second kind and Eulerian numbers.
The main idea is to utilize Newton's interpolation formula
combined with hockey-stick family identities to
derive formulas for multifold sums of powers.
In this manuscript, we use Newton's interpolation formula
in forward finite differences.
However, the Newton's formula approach for sums of powers is quite generic,
as such it does not necessary requires to bind to forward difference operator
specifically. Thus, formulas for sums of powers using backward and central
differences can be found
in~\cite{kolosov_2026_18134757, kolosov_2025_18098442}, respectively.
Allow us to start from the definition of multifold sums of powers.
We utilize the recurrence proposed by Donald Knuth in his article
\textit{Johann Faulhaber and sums of powers}, see~\cite{knuth1993johann}
\begin{align*}
  \KnuthRFoldSum{0}{n}{m}   &= n^m \\
  \KnuthRFoldSum{1}{n}{m}   &= \KnuthRFoldSum{0}{1}{m} + \KnuthRFoldSum{0}{2}{m}
  + \cdots + \KnuthRFoldSum{0}{n}{m} \\
  \KnuthRFoldSum{r+1}{n}{m} &= \KnuthRFoldSum{r}{1}{m} + \KnuthRFoldSum{r}{2}{m}
  + \cdots + \KnuthRFoldSum{r}{n}{m}
\end{align*}
Throughout the paper, we utilize the Newton's interpolation
formula~\cite[Lemma V]{newton1850newton} as stated below
\begin{proposition}[Newton's interpolation formula]
  \begin{align*}
    f(x) = \sum_{j=0}^{\infty} \binom{x-a}{j} \Delta^{j} f(a)
  \end{align*}
  where $\Delta^{k} f(a) = \sum_{j=0}^{k} (-1)^{k-j} \binom{k}{j} f(a+j)$
  are $k$-degree forward finite differences of function $f$
  evaluated in point $a$.
\end{proposition}

Therefore, by setting $f(n)=n^m$ yields

\begin{proposition}[Newton's formula for powers]
  \label{prop:newtons-interpolation-formula-for-powers}
  Let $n,m\ge 0$ be integers, and let $t$ be an arbitrary integer. Then
  \begin{align*}
    n^m = \sum_{k=0}^{m} \binom{n-t}{k} \Delta^{k} t^m
  \end{align*}
\end{proposition}

Which is indeed true, because
\begin{align*}
  n^3 &= 0 \tbinom{n-0}{0}  + 1 \tbinom{n-0}{1}  + 6 \tbinom{n-0}{2}  + 6 \tbinom{n-0}{3} \\
  n^3 &= 1 \tbinom{n-1}{0}  + 7 \tbinom{n-1}{1}  + 12 \tbinom{n-1}{2} + 6 \tbinom{n-1}{3} \\
  n^3 &= 8 \tbinom{n-2}{0}  + 19 \tbinom{n-2}{1} + 18 \tbinom{n-2}{2} + 6 \tbinom{n-2}{3} \\
  n^3 &= 27 \tbinom{n-3}{0} + 37 \tbinom{n-3}{1} + 24 \tbinom{n-3}{2} + 6 \tbinom{n-3}{3}
\end{align*}

\begin{proposition}[Generalized hockey-stick identity]
  \label{prop:generalized-hockey-stick-identity}
  For integers $a,b$ and $j$.
  \begin{align*}
    \sum_{k=a}^{b} \binom{k}{j} = \binom{b+1}{j+1} - \binom{a}{j+1}
  \end{align*}
  \begin{proof}
    We have
    \begin{align*}
      \tsum_{k=a}^{b} \tbinom{k}{j} = \tbinom{a}{j} + \tbinom{a+1}{j}
      + \cdots + \tbinom{b}{j}
    \end{align*}
    which implies
    \begin{align*}
      \tsum_{k=a}^{b} \tbinom{k}{j} =
      \left( \tsum_{k=0}^{b} \tbinom{k}{j} \right)
      -
      \left(\tsum_{k=0}^{a-1} \tbinom{k}{j}\right)
    \end{align*}
    By hockey stick identity
    $\tsum_{k=0}^{n} \binom{k}{j} = \binom{n+1}{j+1}$,
    we get
    \begin{align*}
      \tsum_{k=a}^{b} \tbinom{k}{j}
      =
      \left(
        \tsum_{k=0}^{b} \tbinom{k}{j}
      \right)
      -
      \left(
        \tsum_{k=0}^{a-1} \tbinom{k}{j}
      \right)
      = \tbinom{b+1}{j+1} - \tbinom{a}{j+1}
    \end{align*}
    This completes the proof.
  \end{proof}
\end{proposition}

Therefore,

\begin{lemma} [Shifted hockey stick identity]
  For integers $n,t,j,r$
  \label{lem:shifted-generalized-hockey-stick-identity}
  \begin{align*}
    \sum_{k=1}^{n} \binom{k-t+r}{j+r} = \sum_{k=1-t+r}^{n-t+r} \binom{k}{j+r}
    = \binom{n-t+r+1}{j+r+1} - \binom{1-t+r}{j+r+1}
  \end{align*}
\end{lemma}

Recall the upper negation property of binomial coefficients

\begin{lemma}[Upper negation] For integers $r,k$
  \label{lem:upper-negation}
  \begin{align*}
    \binom{r}{k} = (-1)^{k} \binom{k-r-1}{k}
  \end{align*}
\end{lemma}

Thus,
\begin{align*}
  \binom{1-t+r}{j+r+1}
  &= (-1)^{j+r+1} \binom{(j+r+1)-(1-t+r)-1}{j+r+1} \\
  &= (-1)^{j+r+1} \binom{j+t-1}{j+r+1}
\end{align*}
Which implies negated version of shifted hockey stick
identity~\eqref{lem:shifted-generalized-hockey-stick-identity}.

\begin{lemma} [Negation of shifted hockey stick identity]
  \label{lem:negation-of-shifted-hockey-stick-identity}
  For integers $n,t,j,r$
  \begin{align*}
    \sum_{k=1}^{n} \binom{k-t+r}{j+r}
    = \binom{n-t+r+1}{j+r+1} + (-1)^{j+r} \binom{j+t-1}{j+r+1}
  \end{align*}
\end{lemma}

Hence, by Newton's formula for
powers~\eqref{prop:newtons-interpolation-formula-for-powers},
the ordinary sum of powers is
\begin{align*}
  \KnuthRFoldSum{1}{n}{m}
  = \sum_{k=1}^{n} \sum_{j=0}^{m} \binom{k-t}{j} \Delta^{j} t^m
  = \sum_{j=0}^{m} \Delta^{j} t^m  \sum_{k=1}^{n} \binom{k-t}{j}
\end{align*}

where $t$ is an arbitrary integer.
By shifted hockey stick
identity~\eqref{lem:shifted-generalized-hockey-stick-identity},
the closed form of ordinary sums of powers yields

\begin{proposition}[Ordinary power sums]
  \label{prop:ordinary-sums-of-powers-via-newtons-formula}
  \begin{mdframed}
    For integers $n\geq 0$, and $m\geq 0$, and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{1}{n}{m} = \sum_{j=0}^{m} \Delta^{j} t^m
      \left[ \binom{n-t+1}{j+1} - \binom{1-t}{j+1} \right]
    \end{align*}
  \end{mdframed}
\end{proposition}

For example,
\begin{align*}
  \KnuthRFoldSum{1}{n}{3}
  &= 0\!\left[\tbinom{n+1}{1}-\tbinom{1}{1}\right]
  + 1\!\left[\tbinom{n+1}{2}-\tbinom{1}{2}\right]
  + 6\!\left[\tbinom{n+1}{3}-\tbinom{1}{3}\right]
  + 6\!\left[\tbinom{n+1}{4}-\tbinom{1}{4}\right] \\
  %
  \KnuthRFoldSum{1}{n}{3}
  &= 1\!\left[\tbinom{n}{2}-\tbinom{0}{2}\right]
  + 7\!\left[\tbinom{n}{3}-\tbinom{0}{3}\right]
  + 12\!\left[\tbinom{n}{4}-\tbinom{0}{4}\right]
  + 6\!\left[\tbinom{n}{5}-\tbinom{0}{5}\right] \\
  %
  \KnuthRFoldSum{1}{n}{3}
  &= 8\!\left[\tbinom{n-1}{2}-\tbinom{-1}{2}\right]
  + 19\!\left[\tbinom{n-1}{3}-\tbinom{-1}{3}\right]
  + 18\!\left[\tbinom{n-1}{4}-\tbinom{-1}{4}\right]
  + 6\!\left[\tbinom{n-1}{5}-\tbinom{-1}{5}\right] \\
  %
  \KnuthRFoldSum{1}{n}{3}
  &= 27\!\left[\tbinom{n-2}{2}-\tbinom{-2}{2}\right]
  + 37\!\left[\tbinom{n-2}{3}-\tbinom{-2}{3}\right]
  + 24\!\left[\tbinom{n-2}{4}-\tbinom{-2}{4}\right]
  + 6\!\left[\tbinom{n-2}{5}-\tbinom{-2}{5}\right].
\end{align*}
By negation of shifted hockey stick
identity~\eqref{lem:negation-of-shifted-hockey-stick-identity} yields

\begin{proposition}[Negated ordinary power sums]
  \label{prop:negation-of-ordinary-sums-of-powers-via-newtons-formula}
  \begin{mdframed}
    For integers $n\geq 0$, and $m\geq 0$, and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{1}{n}{m} = \sum_{j=0}^{m} \Delta^{j} t^m
      \left[ \binom{n-t+1}{j+1} + (-1)^j \binom{j+t-1}{j+1} \right]
    \end{align*}
  \end{mdframed}
\end{proposition}
For example,
\begin{align*}
  \KnuthRFoldSum{1}{n}{3}
  &= 0\!\left[\tbinom{n+1}{1} + \tbinom{-1}{1}\right]
  +  1\!\left[\tbinom{n+1}{2} + \tbinom{-1}{2}\right]
  +  6\!\left[\tbinom{n+1}{3} - \tbinom{0}{3}\right]
  +  6\!\left[\tbinom{n+1}{4} + \tbinom{1}{4}\right] \\
  %
  \KnuthRFoldSum{1}{n}{3}
  &= 1\!\left[\tbinom{n}{2} + \tbinom{0}{2}\right]
  + 7\!\left[\tbinom{n}{3} - \tbinom{1}{3}\right]
  + 12\!\left[\tbinom{n}{4} + \tbinom{2}{4}\right]
  + 6\!\left[\tbinom{n}{5} - \tbinom{3}{5}\right] \\
  %
  \KnuthRFoldSum{1}{n}{3}
  &= 8\!\left[\tbinom{n-1}{2} + \tbinom{1}{2}\right]
  + 19\!\left[\tbinom{n-1}{3} - \tbinom{2}{3}\right]
  + 18\!\left[\tbinom{n-1}{4} + \tbinom{3}{4}\right]
  + 6\!\left[\tbinom{n-1}{5} - \tbinom{4}{5}\right] \\
  %
  \KnuthRFoldSum{1}{n}{3}
  &= 27\!\left[\tbinom{n-2}{2} + \tbinom{2}{2}\right]
  + 37\!\left[\tbinom{n-2}{3} - \tbinom{3}{3}\right]
  + 24\!\left[\tbinom{n-2}{4} + \tbinom{4}{4}\right]
  + 6\!\left[\tbinom{n-2}{5} - \tbinom{5}{5}\right]
\end{align*}
% The special cases for $t=0$ and $t=1$ are widely known and appear
% in literature quite frequently

\begin{corollary}
  By setting $t=0$ into~\eqref{prop:ordinary-sums-of-powers-via-newtons-formula}
  \begin{align*}
    \KnuthRFoldSum{1}{n}{m} = \sum_{j=0}^{m} \binom{n+1}{j+1} \Delta^{j} 0^m
  \end{align*}
\end{corollary}

For example,
\begin{align*}
  \KnuthRFoldSum{1}{n}{2} &= 0 \tbinom{n+1}{1} + 1 \tbinom{n+1}{2}
  + 2  \tbinom{n+1}{3}
  \\
  \KnuthRFoldSum{1}{n}{3} &= 0 \tbinom{n+1}{1} + 1 \tbinom{n+1}{2}
  + 6  \tbinom{n+1}{3} + 6   \tbinom{n+1}{4}
  \\
  \KnuthRFoldSum{1}{n}{4} &= 0 \tbinom{n+1}{1} + 1 \tbinom{n+1}{2}
  + 14 \tbinom{n+1}{3} + 36  \tbinom{n+1}{4} + 24  \tbinom{n+1}{5}
  \\
  \KnuthRFoldSum{1}{n}{5} &= 0 \tbinom{n+1}{1} + 1 \tbinom{n+1}{2}
  + 30 \tbinom{n+1}{3} + 150 \tbinom{n+1}{4} + 240 \tbinom{n+1}{5}
  + 120 \tbinom{n+1}{6}
\end{align*}
Note the famous identity
$\KnuthRFoldSum{1}{n}{3} = 0 \tbinom{n+1}{1} + 1 \tbinom{n+1}{2}
+ 6 \tbinom{n+1}{3} + 6 \tbinom{n+1}{4}$
which is  discussed in literature frequently, see~\cite[p. 190]{graham1994concrete}
and in~\cite{pfaff2007deriving}.

The coefficients $0,1,2,0, 1, 6, 6, 0, 1, 14, 36, 24, \dots$ is the sequence
\href{https://oeis.org/A131689}{\texttt{A131689}}
in the OEIS~\cite{sloane2003line}.

\begin{corollary}
  By setting $t=1$ into~\eqref{prop:ordinary-sums-of-powers-via-newtons-formula}
  \begin{align*}
    \KnuthRFoldSum{1}{n}{m} = \sum_{j=0}^{m} \binom{n}{j+1} \Delta^{j} 1^m
  \end{align*}
\end{corollary}
For example,
\begin{align*}
  \KnuthRFoldSum{1}{n}{2} &= 1 \tbinom{n}{1} + 3 \tbinom{n}{2}
  +2   \tbinom{n}{3}
  \\
  \KnuthRFoldSum{1}{n}{3} &= 1 \tbinom{n}{1} + 7 \tbinom{n}{2}
  +12  \tbinom{n}{3} + 6  \tbinom{n}{4}
  \\
  \KnuthRFoldSum{1}{n}{4} &= 1 \tbinom{n}{1} +15 \tbinom{n}{2}
  +50  \tbinom{n}{3} +60  \tbinom{n}{4} + 24 \tbinom{n}{5}
  \\
  \KnuthRFoldSum{1}{n}{5} &= 1 \tbinom{n}{1} +31 \tbinom{n}{2}
  +180 \tbinom{n}{3} +390 \tbinom{n}{4} + 360 \tbinom{n}{5}
  + 120 \tbinom{n}{6}
\end{align*}
The coefficients $1,3,2,1,7,12,6,1,15, \dots$ is the sequence
\href{https://oeis.org/A028246}{\texttt{A028246}}
in the OEIS~\cite{sloane2003line}.
Interestingly enough that the paper~\cite{cereceda2022sums}
give formulas for sums of powers
in terms of generalized Stirling numbers of
the second kind $\stirlingii{k}{j}_{r}$
\begin{theorem}[Cereceda (2022)]
  \begin{align}
    \KnuthRFoldSum{1}{n}{k} &= \sum_{j=0}^{k} j!
    \left[
      \binom{n+1-r}{j+1} + (-1)^j \binom{r+j-1}{j+1}
    \right] \stirlingii{k}{j}_{r}
    \label{eq:cereceda-sums-1}
    \\
    \KnuthRFoldSum{1}{n}{k} &= \sum_{j=0}^{k} j!
    \left[
      \binom{n+1-r}{j+1} + \binom{r+1}{j+1}
    \right] \stirlingii{k}{j}_{-r}
    \label{eq:cereceda-sums-2}
  \end{align}
\end{theorem}
The formula~\eqref{eq:cereceda-sums-1}
is generalized Stirling numbers version of negated ordinary
sums of powers via Newton's
formula~\eqref{prop:negation-of-ordinary-sums-of-powers-via-newtons-formula},
which implies that finite forward differences of powers can be expressed
in terms of generalized Stirling numbers of the second kind:
$\Delta^{j} t^m = j! \stirlingii{m}{j}_{t}$.
The formula~\eqref{eq:cereceda-sums-2} is the special case
of ordinary sums of powers via Newton's
formula~\eqref{prop:ordinary-sums-of-powers-via-newtons-formula} with $t=-r$.
By setting $t=4$ into
ordinary sums of powers via Newton's
formula~\eqref{prop:ordinary-sums-of-powers-via-newtons-formula},
we may observe rather unexpected formulas for sums of powers, namely
\begin{align*}
  \KnuthRFoldSum{1}{n}{0} &= 1  \left( \tbinom{n-3}{1}
  + \tbinom{3}{1}  \right) \\
  \KnuthRFoldSum{1}{n}{1} &= 4  \left( \tbinom{n-3}{1}
  + \tbinom{3}{1}  \right)  + 1  \left( \tbinom{n-3}{2}
  - \tbinom{4}{2}  \right) \\
  \KnuthRFoldSum{1}{n}{2} &= 16 \left( \tbinom{n-3}{1}
  + \tbinom{3}{1}  \right)  + 9  \left( \tbinom{n-3}{2}
  - \tbinom{4}{2}  \right)
  + 2  \left( \tbinom{n-2}{3} + \tbinom{5}{3}  \right) \\
  \KnuthRFoldSum{1}{n}{3} &= 64 \left( \tbinom{n-3}{1}
  + \tbinom{3}{1}  \right)  + 61 \left( \tbinom{n-3}{2}
  - \tbinom{4}{2}  \right)
  + 30 \left( \tbinom{n-3}{3} + \tbinom{5}{3}  \right)
  + 6 \left( \tbinom{n-3}{4} - \tbinom{6}{4}  \right)
\end{align*}
The coefficients $1,4,1,16,9,\dots$ is the sequence
\href{https://oeis.org/A391633}{\texttt{A391633}}
in the OEIS~\cite{sloane2003line}.
In general,
\begin{align*}
  \KnuthRFoldSum{1}{n}{m} = \sum_{j=0}^{m} \Delta^{j} 4^m
  \left[
    \binom{n-3}{j+1} + (-1)^j \binom{j+3}{j+1}
  \right].
\end{align*}

To derive formula for double sum of powers,
we apply summation operator over the ordinary
sum of powers~\eqref{prop:ordinary-sums-of-powers-via-newtons-formula} again
\begin{align*}
  \KnuthRFoldSum{2}{n}{m} = \sum_{j=0}^{m} \Delta^{j} t^m
  \left[ \left( \sum_{k=1}^{n} \binom{k-t+1}{j+1} \right) -
    \binom{1-t}{j+1} \sum_{k=1}^{n} 1
  \right]
\end{align*}
which yields
\begin{align*}
  \KnuthRFoldSum{2}{n}{m}
  = \sum_{j=0}^{m} \Delta^{j} t^m
  \left[
    \left( \sum_{k=1}^{n} \binom{k-t+1}{j+1} \right) - \binom{1-t}{j+1} n
  \right]
\end{align*}
By applying shifted hockey stick
identity~\eqref{lem:shifted-generalized-hockey-stick-identity}, we get

\begin{proposition}[Double sums of powers]
  \label{prop:double-sums-of-powers-via-newtons-formula}
  \begin{mdframed}
    For integers $n\geq 0$, and $m\geq 0$, and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{2}{n}{m}
      = \sum_{j=0}^{m} \Delta^{j} t^m
      \left[
        \binom{n-t+2}{j+1} - \binom{1-t}{j+1} n - \binom{2-t}{j+2}
      \right]
    \end{align*}
  \end{mdframed}

\end{proposition}

By negated hockey stick
identity~\eqref{lem:negation-of-shifted-hockey-stick-identity} yields

\begin{proposition}[Negated double sums of powers]
  \label{prop:negated-double-sums-of-powers-via-newtons-formula}
  \begin{mdframed}
    For integers $n\geq 0$, and $m\geq 0$, and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{2}{n}{m}
      = \sum_{j=0}^{m} \Delta^{j} t^m
      \left[
        \binom{n-t+2}{j+1} + (-1)^j \binom{j+t-1}{j+1} n
        + (-1)^{j+1} \binom{j+t-1}{j+2}
      \right]
    \end{align*}
  \end{mdframed}
\end{proposition}

For example, by setting $t=5$ into double sums of powers via Newton's
interpolation formula~\eqref{prop:double-sums-of-powers-via-newtons-formula},
we get
\begin{align*}
  \KnuthRFoldSum{2}{n}{0}
  &= 1 \left( \tbinom{n-3}{2} + \tbinom{4}{1} n - \tbinom{4}{2} \right)
  \\
  \KnuthRFoldSum{2}{n}{1}
  &= 5 \left( \tbinom{n-3}{2} + \tbinom{4}{1} n - \tbinom{4}{2} \right)
  + 1 \left( \tbinom{n-3}{3} - \tbinom{5}{2} n + \tbinom{5}{3} \right)
  \\
  \KnuthRFoldSum{2}{n}{2}
  &= 25 \left( \tbinom{n-3}{2} + \tbinom{4}{1} n - \tbinom{4}{2} \right)
  + 11 \left( \tbinom{n-3}{3} - \tbinom{5}{2} n + \tbinom{5}{3} \right)
  + 2 \left( \tbinom{n-3}{4} + \tbinom{6}{3} n - \tbinom{6}{4} \right)
\end{align*}
The coefficients $1,5,1,25,11,2,\dots$ is the sequence
\href{https://oeis.org/A391635}{\texttt{A391635}}
in the OEIS~\cite{sloane2003line}.

In general,
\begin{align*}
  \KnuthRFoldSum{2}{n}{m}
  = \sum_{j=0}^{m} \Delta^{j} 5^{m} \left[ \binom{n-3}{j+2} + (-1)^{j} \binom{j+4}{j+1} n^1 + (-1)^{j+1} \binom{j+4}{j+2} n^0 \right]
\end{align*}

In a similar manner, we are able to derive formula for triple sums of powers,
by summing up double sums of
powers~\eqref{prop:double-sums-of-powers-via-newtons-formula},
thus
\begin{align*}
  \KnuthRFoldSum{3}{n}{m}
  = \sum_{j=0}^{m} \Delta^{j} t^m \sum_{k=1}^{n}
  \left[
    \binom{k-t+2}{j+1} - \binom{1-t}{j+1} k^1 - \binom{2-t}{j+2} k^0
  \right]
\end{align*}
Note that $\sum_{k=1}^{n} k^1
= \KnuthRFoldSum{2}{n}{0} $ and $\sum_{k=1}^{n} k^0 = \KnuthRFoldSum{1}{n}{0}$
and $1 = \KnuthRFoldSum{0}{n}{0}$.
Therefore, by shifted hockey stick
identity~\eqref{lem:shifted-generalized-hockey-stick-identity}, we get

\begin{proposition}[Triple sums of powers]
  \label{prop:triple-sums-of-powers-via-newtons-formula}
  \begin{mdframed}
    For integers $n\geq 0$, and $m\geq 0$, and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{3}{n}{m}
      = \sum_{j=0}^{m} \Delta^{j} t^m
      \left[ \binom{n-t+2}{j+1}
        - \binom{1-t}{j+1} \KnuthRFoldSum{2}{n}{0}
        - \binom{2-t}{j+2} \KnuthRFoldSum{1}{n}{0}
        - \binom{3-t}{j+3} \KnuthRFoldSum{0}{n}{0}
      \right]
    \end{align*}
  \end{mdframed}
\end{proposition}

By negated hockey stick
identity~\eqref{lem:negation-of-shifted-hockey-stick-identity} yields

\begin{proposition}[Negated triple sums of powers]
  \begin{mdframed}
    For integers $n\geq 0$, and $m\geq 0$, and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{3}{n}{m} = \sum_{j=0}^{m} \Delta^{j} t^m
      \Bigg[ \binom{n-t+3}{j+3}
        &+ (-1)^{j} \binom{j+t-1}{j+1} \KnuthRFoldSum{2}{n}{0}
        + (-1)^{j+1} \binom{j+t-1}{j+2} \KnuthRFoldSum{1}{n}{0} \\ \nonumber
        &+ (-1)^{j+2} \binom{j+t-1}{j+3} \KnuthRFoldSum{0}{n}{0}
      \Bigg] \nonumber
    \end{align*}
  \end{mdframed}
\end{proposition}

For example, by setting $t=4$ into triple sums of powers via Newton's
formula~\eqref{prop:triple-sums-of-powers-via-newtons-formula},
for $m=1$ we have
\begin{align*}
  \KnuthRFoldSum{3}{n}{1}
  &= 4 \left(
    \tbinom{n-1}{3}
    + \tbinom{3}{1} \KnuthRFoldSum{2}{n}{0}
    - \tbinom{3}{2} \KnuthRFoldSum{1}{n}{0}
    + \tbinom{3}{3} \KnuthRFoldSum{0}{n}{0}
  \right) \\
  &+ 1 \left(
    \tbinom{n-1}{4}
    - \tbinom{4}{2} \KnuthRFoldSum{2}{n}{0}
    + \tbinom{4}{3} \KnuthRFoldSum{1}{n}{0}
  - \tbinom{4}{4} \KnuthRFoldSum{0}{n}{0} \right)
\end{align*}
For $m=2$ we have
\begin{align*}
  \KnuthRFoldSum{3}{n}{2}
  &= 16
  \left(
    \tbinom{n-1}{3}
    + \tbinom{3}{1} \KnuthRFoldSum{2}{n}{0}
    - \tbinom{3}{2} \KnuthRFoldSum{1}{n}{0}
  + \tbinom{3}{3} \KnuthRFoldSum{0}{n}{0} \right) \\
  &+ 9
  \left(
    \tbinom{n-1}{4}
    - \tbinom{4}{2} \KnuthRFoldSum{2}{n}{0}
    + \tbinom{4}{3} \KnuthRFoldSum{1}{n}{0}
    - \tbinom{4}{4} \KnuthRFoldSum{0}{n}{0}
  \right) \\
  &+ 2
  \left( \tbinom{n-1}{5}
    + \tbinom{5}{3} \KnuthRFoldSum{2}{n}{0}
    - \tbinom{5}{4} \KnuthRFoldSum{1}{n}{0}
  + \tbinom{5}{5} \KnuthRFoldSum{0}{n}{0} \right)
\end{align*}
The coefficients $1,4,1,16,9,\dots$ is the sequence
\href{https://oeis.org/A391633}{\texttt{A391633}}
in the OEIS~\cite{sloane2003line}.
In general,
\begin{align*}
  \KnuthRFoldSum{3}{n}{m}
  = \sum_{j=0}^{m} \Delta^{j} 4^m
  \Bigg[
    \binom{n-1}{j+3}
    + (-1)^{j} \binom{j+3}{j+1} \KnuthRFoldSum{2}{n}{0}
    &+ (-1)^{j+1} \binom{j+3}{j+2} \KnuthRFoldSum{1}{n}{0} \\
    &+ (-1)^{j+2} \binom{j+3}{j+3} \KnuthRFoldSum{0}{n}{0}
  \Bigg]
\end{align*}

By repeated application of shifted hockey stick
identity~\eqref{lem:shifted-generalized-hockey-stick-identity},
and Newton's formula for
powers~\eqref{prop:newtons-interpolation-formula-for-powers},
we get formula for multifold sums of powers

\begin{theorem}[Multifold sums of powers]
  \label{theorem:multifold-sums-of-powers-via-newtons-formula}
  \begin{mdframed}
    For integers $r\geq 0$, and $n\geq 0$, and $m\geq 0$,
    and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{r}{n}{m}
      = \sum_{j=0}^{m} \Delta^{j} t^{m}
      \left[
        \binom{n-t+r}{j+r}
        - \sum_{s=1}^{r} \binom{s-t}{j+s} \KnuthRFoldSum{r-s}{n}{0}
      \right]
    \end{align*}
  \end{mdframed}
\end{theorem}

By negated hockey stick
identity~\eqref{lem:negation-of-shifted-hockey-stick-identity}, another
variation of multifold sums of powers follows

\begin{theorem}[Negated multifold sums of powers]
  \label{theorem:negated-multifold-sums-of-powers-via-newtons-formula}
  \begin{mdframed}
    For integers $r\geq 0$, and $n\geq 0$, and $m\geq 0$,
    and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{r}{n}{m}
      = \sum_{j=0}^{m} \Delta^{j} t^{m}
      \left[
        \binom{n-t+r}{j+r}
        + \sum_{s=1}^{r} (-1)^{j+s-1} \binom{j+t-1}{j+s} \KnuthRFoldSum{r-s}{n}{0}
      \right]
    \end{align*}
  \end{mdframed}
\end{theorem}
In its explicit form
\begin{align*}
  \KnuthRFoldSum{r}{n}{m} = \tsum_{j=0}^{m} \Delta^{j} t^{m}
  \Big[
    \tbinom{n-t+r}{j+r}
    &+ (-1)^{j}   \tbinom{j+t-1}{j+1} \KnuthRFoldSum{r-1}{n}{0}
    + (-1)^{j+1} \tbinom{j+t-1}{j+2} \KnuthRFoldSum{r-2}{n}{0} + \cdots \\
    &+ (-1)^{j+r-1} \tbinom{j+t-1}{j+r} \KnuthRFoldSum{0}{n}{0}
  \Big]
\end{align*}
We may observe that

\begin{proposition}[Multifold sum of zero powers]
  \label{prop:multifold-sum-of-zero-powers}
  \begin{mdframed}
    For integers $r \geq 0$ and $n\geq 1$
    \begin{align*}
      \KnuthRFoldSum{r}{n}{0} = \binom{r+n-1}{r}
    \end{align*}
  \end{mdframed}
  \begin{proof}
    \begin{enumerate}
      \item Let $r=0$, then $\KnuthRFoldSum{0}{n}{0}=n^0=\binom{n-1}{0} = 1$,
        by definition.
      \item Let $r=1$, then
        $\KnuthRFoldSum{1}{n}{0}
        = \sum_{k=1}^{n} \binom{k-1}{0}
        = \sum_{k=1}^{n} 1 = \binom{n}{1}$.
      \item Let $r=2$, then
        $\KnuthRFoldSum{2}{n}{0}= \sum_{k=1}^{n} \binom{k}{1}
        = \sum_{k=1}^{n} k = \binom{n+1}{2}$.
      \item Let $r=3$, then
        $\KnuthRFoldSum{3}{n}{0}
        = \sum_{k=1}^{n} \binom{k+1}{2}
        = \binom{n+2}{3}$.
      \item By induction over $r$ and hockey stick identity
        $\sum_{k=r}^{n} \binom{k}{r} = \binom{n+1}{r+1}$,
        the claim follows
        $\KnuthRFoldSum{r}{n}{0} = \binom{r+n-1}{r}$.
    \end{enumerate}
    This completes the proof.
  \end{proof}
\end{proposition}
Hence,
\begin{align*}
  \KnuthRFoldSum{r-s}{n}{0} = \binom{r-s+n-1}{r-s}
\end{align*}

Which yields binomial variations of
multifold sums of powers via Newton's
formula~\eqref{theorem:multifold-sums-of-powers-via-newtons-formula}

\begin{proposition}[Multifold Binomial sums of powers]
  \label{prop:multifold-sums-of-powers-binomial-form}
  \begin{mdframed}
    For integers $r\geq 0$, and $n\geq 0$, and $m\geq 0$,
    and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{r}{n}{m}
      = \sum_{j=0}^{m} \Delta^{j} t^{m}
      \left[ \binom{n-t+r}{j+r} -
        \sum_{s=1}^{r} \binom{s-t}{j+s} \binom{r-s+n-1}{r-s}
      \right]
    \end{align*}
  \end{mdframed}
\end{proposition}

By negated hockey stick
identity~\eqref{lem:negation-of-shifted-hockey-stick-identity} yields

\begin{proposition}[Negated multifold Binomial sum of powers]
  \label{prop:negated-multifold-sums-of-powers-binomial-form}
  \begin{mdframed}
    For integers $r\geq 0$, and $n\geq 0$, and $m\geq 0$,
    and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{r}{n}{m}
      &= \sum_{j=0}^{m} \Delta^{j} t^{m}
      \left[ \binom{n-t+r}{j+r} +
        \sum_{s=1}^{r} (-1)^{j+s-1} \binom{j+t-1}{j+s} \binom{r-s+n-1}{r-s}
      \right]
    \end{align*}
  \end{mdframed}
\end{proposition}

By re-indexing the sum in $s$ gives

\begin{proposition}[Shifted multifold Binomial sum of powers]
  \label{prop:negated-multifold-sums-of-powers-binomial-form-reindexed}
  \begin{mdframed}
    For integers $r\geq 0$, and $n\geq 0$, and $m\geq 0$,
    and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{r}{n}{m}
      = \sum_{j=0}^{m} \Delta^{j} t^{m}
      \left[ \binom{n-t+r}{j+r} +
        \sum_{s=0}^{r-1} (-1)^{j+s} \binom{j+t-1}{j+s+1} \binom{r-s+n-2}{r-s-1}
      \right]
    \end{align*}
  \end{mdframed}
\end{proposition}
