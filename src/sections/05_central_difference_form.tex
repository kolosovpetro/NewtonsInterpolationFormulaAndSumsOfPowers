The Theorem~\eqref{theorem:multifold-sums-of-powers-via-newtons-formula}
can be expressed in terms of central differences easily, because,
\begin{proposition} For integers $j,m,t$,
  \label{prop:forward-via-central-differences}
  \begin{align*}
    \Delta^{j} t^{m} = \delta^{j} \left( t+ \frac{j}{2} \right)^{m}.
  \end{align*}
\end{proposition}
Thus, by negated multifold sums of
powers~\eqref{theorem:negated-multifold-sums-of-powers-via-newtons-formula}
and by Proposition~\eqref{prop:forward-via-central-differences},
we obtain the formula for multifold sums of powers
in terms of central differences $\delta$.
\begin{proposition}[Multifold central sums of powers]
  \label{prop:multifold-sums-of-powers-via-central-differences}
  For integers $r,n,m \geq 0$, and an arbitrary integer $t$,
  \begin{align*}
    \KnuthRFoldSum{r}{n}{m}
    = \sum_{j=0}^{m}
    \left[\binom{n-t+r}{j+r}
      + \sum_{s=1}^{r} (-1)^{j+s-1} \binom{j+t-1}{j+s} \KnuthRFoldSum{r-s}{n}{0}
    \right] \delta^{j} \left( t + \frac{j}{2} \right)^m.
  \end{align*}
\end{proposition}
