\begin{lemma}[Worpitzky identity]
  \label{lem:worpitzky-identity}
  For non-negative integers $t,m$
  \begin{align*}
    t^{m} = \sum_{k=0}^{m} \binom{t+k}{m} \eulerian{m}{k},
  \end{align*}
  where $\eulerian{n}{k}$ are Eulerian numbers.
  \begin{proof}
    See the (1883) work of J. Worpitzky~\cite{Worpitzky1883}.
  \end{proof}
\end{lemma}
Thus, we obtain formula for finite differences of powers, in terms
of Eulerian numbers $\eulerian{m}{s}$
\begin{lemma}[Eulerian finite differences]
  \label{lem:finite-differences-via-eulerian-numbers}
  For non-negative integers $t,k$, and an arbitrary integer $t$,
  \begin{align*}
    \Delta^{k} t^{m} = \sum_{s=0}^{m} \binom{t+s}{m-k} \eulerian{m}{s}.
  \end{align*}
  \begin{proof}
    By binomial recurrence $\binom{n+1}{k} = \binom{n}{k} + \binom{n}{k-1}$,
    and by Worpitzky identity~\eqref{lem:worpitzky-identity}.
  \end{proof}
\end{lemma}
