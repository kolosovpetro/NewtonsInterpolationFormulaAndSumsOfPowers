We begin with Worpitzky's identity.
\begin{lemma}[Worpitzky Identity]
  \label{lem:worpitzky-identity}
  For non-negative integers $t$ and $m$,
  \begin{align*}
    t^{m} = \sum_{k=0}^{m} \binom{t+k}{m} \eulerian{m}{k},
  \end{align*}
  where $\eulerian{n}{k}$ are Eulerian numbers.
  \begin{proof}
    The identity is given by J. Worpitzky in~\cite{Worpitzky1883}.
  \end{proof}
\end{lemma}
The following lemma expresses finite differences of powers in terms of Eulerian numbers $\eulerian{m}{s}$.
\begin{lemma}[Eulerian finite differences]
  \label{lem:finite-differences-via-eulerian-numbers}
  For non-negative integers $k$ and $m$ and an arbitrary integer $t$,
  \begin{align*}
    \Delta^{k} t^{m} = \sum_{j=0}^{m} \binom{t+j}{m-k} \eulerian{m}{j}.
  \end{align*}
  \begin{proof}
    This follows from the binomial recurrence $\tbinom{n+1}{k} = \tbinom{n}{k} + \tbinom{n}{k-1}$ and Worpitzky's identity in Lemma~\eqref{lem:worpitzky-identity}.
  \end{proof}
\end{lemma}
Substituting this expression into Theorem~\eqref{thm:multifold-sums-of-powers} gives the following formula.
\begin{proposition}[Multifold Eulerian sums]
  \label{prop:multifold-eulerian-sums}
  For non-negative integers $r$, $n$, and $m$ and an arbitrary integer $t$,
  \begin{align*}
    \KnuthRFoldSum{r}{n}{m}
    = \sum_{k=0}^{m} \sum_{j=0}^{m}
    \left[
      \binom{n-t+r}{k+r}
      -
      \sum_{s=0}^{r-1}
      \binom{r-s-t}{k+r-s}
      \KnuthRFoldSum{s}{n}{0}
    \right] \binom{t+j}{m-k} \eulerian{m}{j}.
  \end{align*}
\end{proposition}
Applying Lemma~\eqref{lem:multifold-sum-of-zero-powers} gives the following pure binomial form.
\begin{proposition}[Multifold Eulerian-Binomial sums]
  \label{prop:multifold-eulerian-binomial-sums}
  For non-negative integers $r$, $n$, and $m$ and an arbitrary integer $t$,
  \begin{align*}
    \KnuthRFoldSum{r}{n}{m}
    = \sum_{k=0}^{m} \sum_{j=0}^{m}
    \left[
      \binom{n-t+r}{k+r}
      -
      \sum_{s=0}^{r-1}
      \binom{r-s-t}{k+r-s}
      \binom{s+n-1}{s}
    \right] \binom{t+j}{m-k} \eulerian{m}{j}.
  \end{align*}
\end{proposition}
