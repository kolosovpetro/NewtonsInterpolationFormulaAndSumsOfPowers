In addition, we are able to express multifold sums of powers
via Eulerian numbers, by expressing the forward finite difference
via the Worpitzky identity~\cite{Worpitzky1883}

\begin{lemma}[Worpitzky identity]
  For non-negative integers $t,m$
  \begin{align*}
    t^{m} = \sum_{k=0}^{m} \binom{t+k}{m} \eulerian{m}{k}
  \end{align*}
\end{lemma}

where $\eulerian{n}{k}$ are Eulerian numbers.

Thus,

\begin{lemma}[Finite difference via Eulerian numbers]
  \label{lem:finite-differences-via-eulerian-numbers}
  For non-negative integers $j,m$ and an arbitrary integer $t$
  \begin{align*}
    \Delta^{j} t^{m} = \sum_{k=0}^{m} \binom{t+k}{m-j} \eulerian{m}{k}
  \end{align*}
\end{lemma}

Hence, by multifold sums of powers via Newton's
formula~\eqref{theorem:multifold-sums-of-powers-via-newtons-formula}
and forward finite difference via Eulerian
numbers~\eqref{lem:finite-differences-via-eulerian-numbers}

\begin{proposition}[Multifold sums of powers via Eulerian numbers]
  \begin{mdframed}
    For non-negative integers $r,n,m$ and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{r}{n}{m}
      = \sum_{j=0}^{m} \sum_{k=0}^{m}
      \left[ \binom{n-t+r}{j+r} - \sum_{s=1}^{r} \binom{s-t}{j+s}
      \KnuthRFoldSum{r-s}{n}{0} \right]
      \binom{t+k}{m-j} \eulerian{m}{k}
    \end{align*}
  \end{mdframed}
\end{proposition}

Therefore, by negated multifold sums of powers via Newton's
formula~\eqref{theorem:negated-multifold-sums-of-powers-via-newtons-formula}
and forward finite difference via Eulerian
numbers~\eqref{lem:finite-differences-via-eulerian-numbers}
\begin{proposition}[Negated multifold sums of powers via Eulerian numbers]
  \label{prop:multifold-sums-of-powers-via-eulerian-numbers}
  \begin{mdframed}
    For non-negative integers $r,n,m$ and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{r}{n}{m}
      =\sum_{j=0}^{m} \sum_{k=0}^{m}
      \left[ \binom{n-t+r}{j+r} + \sum_{s=1}^{r} (-1)^{j+s-1}
      \binom{j+t-1}{j+s} \KnuthRFoldSum{r-s}{n}{0} \right]
      \binom{t+k}{m-j} \eulerian{m}{k}
    \end{align*}
  \end{mdframed}
\end{proposition}

Which implies variations for ordinary sums of powers
via Newton's formula~\eqref{prop:ordinary-sums-of-powers-via-newtons-formula}

\begin{proposition}[Negated ordinary sums of powers via Eulerian numbers]
  \label{prop:sums-of-powers-via-eulerian-numbers}
  \begin{mdframed}
    For non-negative integers $n,m$ and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{1}{n}{m}
      = \sum_{j=0}^{m} \sum_{k=0}^{m}
      \left[ \binom{n-t+1}{j+1} + (-1)^j \binom{j+t-1}{j+1} \right]
      \binom{t+k}{m-j} \eulerian{m}{k}
    \end{align*}
  \end{mdframed}
\end{proposition}

By setting $t=0$ into
ordinary sums of powers via Eulerian
numbers~\eqref{prop:sums-of-powers-via-eulerian-numbers}
yields a well-known identity for
sums of powers in terms of Eulerian numbers
\begin{corollary}[Sums of powers via Eulerian numbers in zero]
  \begin{mdframed}
    For non-negative integers $n,m$
    \begin{align*}
      \KnuthRFoldSum{1}{n}{m}=
      \sum_{j=0}^{m} \sum_{k=0}^{m}
      \binom{n+1}{j+1} \binom{k}{m-j} \eulerian{m}{k}
    \end{align*}
  \end{mdframed}
\end{corollary}

By~\eqref{prop:multifold-sums-of-powers-binomial-form} and
by~\eqref{prop:negated-multifold-sums-of-powers-binomial-form}
yield multifold sums of powers in Eulerian-Binomial form

\begin{proposition}[Multifold sums of powers in Eulerian-Binomial form]
  % \label{prop:multifold-sums-of-powers-binomial-form}
  \begin{mdframed}
    For non negative integers $r,n,m$ and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{r}{n}{m}
      = \sum_{j=0}^{m} \sum_{k=0}^{m}
      \left[ \binom{n-t+r}{j+r} -
        \sum_{s=1}^{r} \binom{s-t}{j+s} \binom{r-s+n-1}{r-s}
      \right]
      \binom{t+k}{m-j} \eulerian{m}{k}
    \end{align*}
  \end{mdframed}
\end{proposition}

\begin{proposition}[Negated multifold sums of powers in Eulerian-Binomial form]
  % \label{prop:negated-multifold-sums-of-powers-binomial-form}
  \begin{mdframed}
    For non-negative integers $r,n,m$ and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{r}{n}{m}
      = \sum_{j=0}^{m} \sum_{k=0}^{m}
      \left[ \tbinom{n-t+r}{j+r} +
        \sum_{s=1}^{r} (-1)^{j+s-1} \tbinom{j+t-1}{j+s} \tbinom{r-s+n-1}{r-s}
      \right]
      \tbinom{t+k}{m-j} \teulerian{m}{k}
    \end{align*}
  \end{mdframed}
\end{proposition}
