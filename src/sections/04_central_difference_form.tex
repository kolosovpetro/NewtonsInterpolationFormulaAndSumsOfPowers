The formula for multifold sums of powers via Newton's series~\eqref{theorem:multifold-sums-of-powers-via-newtons-series}
can be altered to be in terms of central differences easily, because
\begin{proposition} For integers $j,m,t$
    \label{prop:forward-via-central-differences}
    \begin{align*}
        \Delta^{j} t^{m} = \delta^{j} \left( t+ \frac{j}{2} \right)^{m}
    \end{align*}
\end{proposition}
Thus,
\begin{proposition}[Multifold sums of powers via central differences]
    \label{prop:multifold-sums-of-powers-via-central-differences}
    \begin{mdframed}
        For non-negative integers $r,n,m$ and an arbitrary integer $t$
        \begin{align*}
            \KnuthRFoldSum{r}{n}{m} = \sum_{j=0}^{m} \delta^{j} \left( t + \frac{j}{2} \right)^m \left[ \left( \sum_{s=1}^{r} (-1)^{j+s-1} \binom{j+t-1}{j+s} \KnuthRFoldSum{r-s}{n}{0} \right) + \binom{n-t+r}{j+r} \right]
        \end{align*}
    \end{mdframed}
    \begin{proof}
        By multifold sums of powers via Newton's series~\eqref{theorem:multifold-sums-of-powers-via-newtons-series} and
        by proposition~\eqref{prop:forward-via-central-differences}.
    \end{proof}
\end{proposition}
