The formula for multifold sums of powers via Newton's
series~\eqref{theorem:multifold-sums-of-powers-via-newtons-series}
can be expressed in terms of backward differences easily, because
\begin{proposition} For integers $j,m,t$
  \label{prop:forward-via-backward-differences}
  \begin{align*}
    \Delta^{j} t^{m} = \nabla^{j} (t+j)^{m}
  \end{align*}
\end{proposition}
Thus,
\begin{proposition}[Multifold sums of powers via backward differences]
  \label{prop:multifold-sums-of-powers-via-backward-difference}
  \begin{mdframed}
    For non-negative integers $r,n,m$ and an arbitrary integer $t$
    \begin{align*}
      \KnuthRFoldSum{r}{n}{m}
      = \sum_{j=0}^{m} \nabla^{j} (t+j)^{m} \left[ \binom{n-t+r}{j+r} +  \sum_{s=1}^{r} (-1)^{j+s-1} \binom{j+t-1}{j+s} \KnuthRFoldSum{r-s}{n}{0} \right]
    \end{align*}
  \end{mdframed}
  \begin{proof}
    By multifold sums of powers via Newton's
    series~\eqref{theorem:multifold-sums-of-powers-via-newtons-series} and
    by proposition~\eqref{prop:forward-via-backward-differences}.
  \end{proof}
\end{proposition}
